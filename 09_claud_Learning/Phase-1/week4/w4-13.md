### LC #417 — Pacific Atlantic Water Flow ✅

```cpp
void bfs(vector<vector<int>>& heights, vector<vector<bool>>& visited, queue<pair<int,int>> q) {
    int rows = heights.size();
    int cols = heights[0].size();
    while (!q.empty()) {
        auto [cx, cy] = q.front();
        q.pop();
        if (cx+1 < rows && !visited[cx+1][cy] && heights[cx+1][cy] >= heights[cx][cy]) { visited[cx+1][cy] = true; q.push({cx+1,cy}); }
        if (cx-1 >= 0  && !visited[cx-1][cy] && heights[cx-1][cy] >= heights[cx][cy]) { visited[cx-1][cy] = true; q.push({cx-1,cy}); }
        if (cy+1 < cols && !visited[cx][cy+1] && heights[cx][cy+1] >= heights[cx][cy]) { visited[cx][cy+1] = true; q.push({cx,cy+1}); }
        if (cy-1 >= 0  && !visited[cx][cy-1] && heights[cx][cy-1] >= heights[cx][cy]) { visited[cx][cy-1] = true; q.push({cx,cy-1}); }
    }
}

vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
    int rows = heights.size(), cols = heights[0].size();
    vector<vector<bool>> pacific(rows, vector<bool>(cols, false));
    vector<vector<bool>> atlantic(rows, vector<bool>(cols, false));
    queue<pair<int,int>> pac, atl;
    for (int i = 0; i < cols; i++) {
        pac.push({0,i}); pacific[0][i] = true;
        atl.push({rows-1,i}); atlantic[rows-1][i] = true;
    }
    for (int i = 0; i < rows; i++) {
        pac.push({i,0}); pacific[i][0] = true;
        atl.push({i,cols-1}); atlantic[i][cols-1] = true;
    }
    bfs(heights, pacific, pac);
    bfs(heights, atlantic, atl);
    vector<vector<int>> result;
    for (int i = 0; i < rows; i++)
        for (int j = 0; j < cols; j++)
            if (pacific[i][j] && atlantic[i][j]) result.push_back({i,j});
    return result;
}
```

**Key:** Reverse thinking — start from ocean borders, go uphill (≥). One BFS function, two grids, two calls. Top/bottom rows seed = `i < cols`. Left/right cols seed = `i < rows`.

### LC #130 — Surrounded Regions ✅

```cpp
void bfs(vector<vector<char>>& board, queue<pair<int,int>>& check) {
    int rows = board.size(), cols = board[0].size();
    while (!check.empty()) {
        auto [cx, cy] = check.front();
        check.pop();
        if (cx+1 < rows && board[cx+1][cy] == 'O') { board[cx+1][cy] = 'T'; check.push({cx+1,cy}); }
        if (cx-1 >= 0  && board[cx-1][cy] == 'O') { board[cx-1][cy] = 'T'; check.push({cx-1,cy}); }
        if (cy+1 < cols && board[cx][cy+1] == 'O') { board[cx][cy+1] = 'T'; check.push({cx,cy+1}); }
        if (cy-1 >= 0  && board[cx][cy-1] == 'O') { board[cx][cy-1] = 'T'; check.push({cx,cy-1}); }
    }
}

void solve(vector<vector<char>>& board) {
    int rows = board.size(), cols = board[0].size();
    queue<pair<int,int>> check;
    for (int i = 0; i < cols; i++) {
        if (board[0][i] == 'O')      { check.push({0,i});      board[0][i] = 'T'; }
        if (board[rows-1][i] == 'O') { check.push({rows-1,i}); board[rows-1][i] = 'T'; }
    }
    for (int i = 0; i < rows; i++) {
        if (board[i][0] == 'O')      { check.push({i,0});      board[i][0] = 'T'; }
        if (board[i][cols-1] == 'O') { check.push({i,cols-1}); board[i][cols-1] = 'T'; }
    }
    bfs(board, check);
    for (auto& p : board)
        for (auto& q : p) {
            if (q == 'O') q = 'X';
            else if (q == 'T') q = 'O';
        }
}
```

**Key:** Save safe O's first (border-connected) by marking 'T'. After BFS: O→X (surrounded), T→O (safe restore). Grid itself is the visited marker — no separate bool grid needed.

### Recursion on trees — 5-step formula

```
1. Base case      → if (root == nullptr) return SOMETHING
2. Go left        → recurse on root->left
3. Go right       → recurse on root->right
4. Combine        → use left and right answers
5. Return         → send answer up
```

```cpp
int maxDepth(TreeNode* root) {
    if (root == nullptr) return 0;
    queue<TreeNode*> q;
    q.push(root);
    int depth = 0;
    while (!q.empty()) {
        int levelSize = q.size();
        for (int i = 0; i < levelSize; i++) {
            TreeNode* node = q.front();
            q.pop();
            if (node->left)  q.push(node->left);
            if (node->right) q.push(node->right);
        }
        depth++;
    }
    return depth;
}
```

### LC #100 — Same Tree recursive ✅

```cpp
bool isSameTree(TreeNode* p, TreeNode* q) {
    if (p == nullptr && q == nullptr) return true;
    else if (p == nullptr || q == nullptr) return false;
    else if (p->val != q->val) return false;
    bool left  = isSameTree(p->left,  q->left);
    bool right = isSameTree(p->right, q->right);
    return left && right;
}
```

### LC #100 — Same Tree iterative ✅

```cpp
bool isSameTree(TreeNode* p, TreeNode* q) {
    stack<pair<TreeNode*, TreeNode*>> st;
    st.push({p, q});
    while (!st.empty()) {
        pair<TreeNode*, TreeNode*> current = st.top();
        st.pop();
        TreeNode* node1 = current.first;
        TreeNode* node2 = current.second;
        if (node1 == nullptr && node2 == nullptr) continue;
        else if (node1 == nullptr || node2 == nullptr) return false;
        else if (node1->val != node2->val) return false;
        st.push({node1->left, node2->left});
        st.push({node1->right, node2->right});
    }
    return true;
}
```

### LC #572 — Subtree of Another Tree ✅

```cpp
bool isSubtree(TreeNode* root, TreeNode* subRoot) {
    if (root == nullptr) return false;
    if (root->val == subRoot->val) {
        if (isSameTree(root, subRoot)) return true;
    }
    bool left  = isSubtree(root->left,  subRoot);
    bool right = isSubtree(root->right, subRoot);
    return left || right;
}
```

**Complexity:** O(n×m)

```cpp
void dfs(vector<vector<char>>& grid, int x, int y) {
    int rows = grid.size();
    int cols = grid[0].size();
    if (x < 0 || x >= rows || y < 0 || y >= cols) return;
    if (grid[x][y] == '0') return;
    grid[x][y] = '0';
    dfs(grid, x+1, y);
    dfs(grid, x-1, y);

int numIslands(vector<vector<char>>& grid) {

        for (int y = 0; y < grid[0].size(); y++)
            if (grid[x][y] == '1') { dfs(grid, x, y); count++; }
    return count;
}
```

### LC #200 — Number of Islands BFS ✅

```cpp
void bfs(vector<vector<char>>& grid, int x, int y) {
    int rows = grid.size(), cols = grid[0].size();
    queue<pair<int,int>> q;
    q.push({x, y});
    grid[x][y] = '0';
    while (!q.empty()) {
        auto [cx, cy] = q.front(); q.pop();
        if (cx+1 < rows && grid[cx+1][cy] == '1') { grid[cx+1][cy] = '0'; q.push({cx+1,cy}); }
        if (cx-1 >= 0  && grid[cx-1][cy] == '1') { grid[cx-1][cy] = '0'; q.push({cx-1,cy}); }
        if (cy+1 < cols && grid[cx][cy+1] == '1') { grid[cx][cy+1] = '0'; q.push({cx,cy+1}); }
        if (cy-1 >= 0  && grid[cx][cy-1] == '1') { grid[cx][cy-1] = '0'; q.push({cx,cy-1}); }
    }
}
```

**Key:** mark '0' immediately when pushing — prevents same cell added twice.

### LC #133 — Clone Graph ✅

```cpp
Node* dfs(Node* node, unordered_map<Node*, Node*>& visited) {
    if (node == nullptr) return nullptr;
    if (visited.count(node)) return visited[node];
    Node* clone = new Node(node->val);
    visited[node] = clone;
    for (Node* p : node->neighbors)
        clone->neighbors.push_back(dfs(p, visited));
    return clone;
}
Node* cloneGraph(Node* node) {
    unordered_map<Node*, Node*> visited;
    return dfs(node, visited);
}
```

**Key:** `visited[node] = clone` before the for loop — prevents infinite loop on cycles.
