## DP — WORD BREAK (LC #139) — COVERED W5 D5

### Recursion
```cpp
bool solve(string& s, unordered_set<string>& word, int i) {
    if (i == s.size()) return true;
    for (int j = i; j < s.size(); j++) {
        string wstr = s.substr(i, j - i + 1);
        if (word.count(wstr)) {
            if (solve(s, word, j + 1)) return true;
        }
    }
    return false;
}

bool wordBreak(string s, vector<string>& wordDict) {
    unordered_set<string> dict(wordDict.begin(), wordDict.end());
    return solve(s, dict, 0);
}
```

### Memoisation
```cpp
bool solve(string& s, unordered_set<string>& word, int i, vector<int>& memo) {
    if (i == s.size()) return true;
    if (memo[i] != -1) return memo[i];
    for (int j = i; j < s.size(); j++) {
        string wstr = s.substr(i, j - i + 1);
        if (word.count(wstr)) {
            if (solve(s, word, j + 1, memo)) {
                return memo[i] = true;
            }
        }
    }
    return memo[i] = false;
}

bool wordBreak(string s, vector<string>& wordDict) {
    vector<int> memo(s.size() + 1, -1);
    unordered_set<string> dict(wordDict.begin(), wordDict.end());
    return solve(s, dict, 0, memo);
}
```

## DP — LCS (LC #1143) — COVERED W5 D3

### Recurrence
```
match:    dp[i][j] = 1 + dp[i-1][j-1]
no match: dp[i][j] = max(dp[i-1][j], dp[i][j-1])
```

### Memoisation
```cpp
int lcsrecursion(string& text1, string& text2, int i, int j, vector<vector<int>>& memo) {
    if (i < 0 || j < 0) return 0;
    if (memo[i][j] != -1) return memo[i][j];
    if (text1[i] == text2[j])
        return memo[i][j] = 1 + lcsrecursion(text1, text2, i-1, j-1, memo);
    return memo[i][j] = max(lcsrecursion(text1, text2, i-1, j, memo),
                            lcsrecursion(text1, text2, i, j-1, memo));
}
```

### Tabulation
```cpp
int lcstabulation(string& text1, string& text2, vector<vector<int>>& table) {
    int m = text1.size(), n = text2.size();
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (text1[i-1] == text2[j-1])
                table[i][j] = 1 + table[i-1][j-1];
            else
                table[i][j] = max(table[i-1][j], table[i][j-1]);
        }
    }
    return table[m][n];
}
```

### O(n) space
```cpp
int lcsprevcurr(string& text1, string& text2) {
    int m = text1.size(), n = text2.size();
    vector<int> prev(n+1, 0), cur(n+1, 0);
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (text1[i-1] == text2[j-1])
                cur[j] = 1 + prev[j-1];
            else
                cur[j] = max(prev[j], cur[j-1]);
        }
        prev = cur;
    }
    return prev[n];
}
```

## DP — EDIT DISTANCE (LC #72) — COVERED W5 D3

### Recurrence
```
match:    dp[i][j] = dp[i-1][j-1]
no match: dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
                               delete      insert       replace
```

### Base cases
- `i < 0` → `j + 1` (insert remaining word2 chars)
- `j < 0` → `i + 1` (delete remaining word1 chars)

### Memoisation
```cpp
int edit(string& w1, string& w2, int i, int j, vector<vector<int>>& memo) {
    if (i < 0) return j + 1;
    if (j < 0) return i + 1;
    if (memo[i][j] != -1) return memo[i][j];
    if (w1[i] == w2[j])
        return memo[i][j] = edit(w1, w2, i-1, j-1, memo);
    return memo[i][j] = 1 + min({edit(w1, w2, i-1, j-1, memo),
                                  edit(w1, w2, i, j-1, memo),
                                  edit(w1, w2, i-1, j, memo)});
}
```

## DP — HOUSE ROBBER (LC #198) — COVERED W5 D2

### Recurrence
```
dp[i] = max(nums[i] + dp[i-2], dp[i-1])
```

### Full solution
```cpp
class Solution {
public:
    int memoisation(vector<int>& nums, int i, vector<int>& memo) {
        if (i == 0) return nums[0];
        if (i == 1) return max(nums[0], nums[1]);
        if (memo[i] != -1) return memo[i];
        memo[i] = max(nums[i] + memoisation(nums, i-2, memo), memoisation(nums, i-1, memo));
        return memo[i];
    }

    int dp(int prev2, int prev1, vector<int>& nums) {
        int current;
        for (int i = 2; i < nums.size(); i++) {
            current = max(nums[i] + prev2, prev1);
            prev2 = prev1;
            prev1 = current;
        }
        return current;
    }

    int rob(vector<int>& nums) {
        if (nums.size() == 1) return nums[0];
        vector<int> memo(nums.size(), -1);
        return dp(nums[0], nums[1], nums);
    }
};
```

## DP — CLIMBING STAIRS (LC #70) — COVERED W5 D1

### Key insight
`climb(n) = climb(n-1) + climb(n-2)` — Fibonacci in disguise.

### Version 3 — O(1) space
```cpp
int memory(int n, int prev1, int prev2) {
    int current = 0;
    for (int i = 3; i <= n; i++) {
        current = prev1 + prev2;
        prev2 = prev1;
        prev1 = current;
    }
    return current;
}
int climbStairs(int n) {
    if (n == 1) return 1;
    if (n == 2) return 2;
    return memory(n, 2, 1);
}
```

## DFS TOPOLOGICAL SORT — COVERED W5 D1

```cpp
bool dfs(int node, vector<int>& state, vector<vector<int>>& adj, vector<int>& result) {
    state[node] = 1;
    for (auto& p : adj[node]) {
        if (state[p] == 1) return false;
        else if (state[p] == 2) continue;
        else if (!dfs(p, state, adj, result)) return false;
    }
    state[node] = 2;
    result.push_back(node);
    return true;
}
```

```cpp
    int mid = lo + (hi - lo) / 2;
    if (nums[mid] == target) return mid;
    else if (nums[mid] < target) lo = mid + 1;
    else hi = mid - 1;
}
return -1;
```
