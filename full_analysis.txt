vector<int> bubblesort(vector<int> &arr)
{
    int count = 0;
    int size = arr.size();
    for (int i = 0; i < size - 1; ++i) // no. of passes
    {
        for (int j = 0; j < (size - i) - 1; ++j)
        {
            if (arr[j] > arr[j + 1])
            {
                count++;
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
        if (count == 1) // so that if already sorted then no passes are used
            count--;
        return arr;
    }
    return arr;
}

int binarysearch(vector<int> &arr, int target)
{
    int l = 0;
    int r = arr.size() - 1;

    int mid = l + (r - l) / 2;
    while (l <= r)
    {
        mid = l + (r - l) / 2;
        if (arr[mid] < target)
        {
            l = mid + 1;
        }
        else
        {
            r = mid - 1;
        }
    }
    return l + 1;
}
